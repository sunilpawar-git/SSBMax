## Deterministic Extraction (PDF → JSON)

**Philosophy: NO LLM for correctness. Parse, don't infer.**

**Bad Approach (Hallucination Risk):**
```python
# ❌ DON'T DO THIS
import requests
response = requests.post("https://api.openai.com/v1/...", 
  json={"image_url": pdf_page})
data = response.json()  # Might hallucinate missing data

# Problems:
# - LLM hallucinates incorrect answer keys
# - Unpredictable outputs (no ground truth)
# - Can't verify correctness without human review
```

**Good Approach (Deterministic):**
```python
import fitz  # PyMuPDF

# 1. Extract text deterministically
pdf_doc = fitz.open("test.pdf")
page = pdf_doc[0]
text = page.get_text("text")  # Deterministic: same PDF → same text

# 2. Parse structure from text layer (no vision model)
questions = parse_questions_from_text(text)

# 3. Extract images (if needed) by position
for image_rect in page.get_images():
  # Get image by position, not by inference
  image_data = page.get_pixmap(clip=image_rect)
  image_data.save(f"image_{image_rect.x0}_{image_rect.y0}.png")

# 4. LLM ONLY for enrichment (tags, difficulty, metadata)
enriched = enrich_with_gemini(questions)

# Result: Correct data + enriched metadata
```

## Batch Validation Before Upload (Quality Gates)

**Pattern: Validate → Preview → Upload**

```javascript
// Step 1: Validate structure
function validateQuestions(questions) {
  const errors = [];
  
  questions.forEach((q, idx) => {
    if (!q.id) errors.push(`Question ${idx}: Missing id`);
    if (!q.questionText) errors.push(`Question ${idx}: Missing text`);
    if (!Array.isArray(q.options) || q.options.length < 2) {
      errors.push(`Question ${idx}: Need ≥2 options`);
    }
    if (!q.correctAnswerId) {
      errors.push(`Question ${idx}: Missing correctAnswerId`);
    }
    // Verify correctAnswerId is valid
    if (!q.options.find(opt => opt.id === q.correctAnswerId)) {
      errors.push(`Question ${idx}: correctAnswerId not in options`);
    }
  });
  
  return { isValid: errors.length === 0, errors };
}

// Step 2: Generate HTML preview
function generatePreview(questions) {
  let html = "<html><body>";
  questions.forEach(q => {
    html += `<h3>${q.questionText}</h3>`;
    q.options.forEach(opt => {
      const correct = opt.id === q.correctAnswerId ? "✓ CORRECT" : "";
      html += `<p>${opt.text} ${correct}</p>`;
    });
  });
  html += "</body></html>";
  return html;
}

// Step 3: Manual review
console.log("Preview: http://localhost:8000/preview.html");
console.log("Review the preview, then run: node upload.js");

// Step 4: Upload only after approval
async function uploadToFirestore(questions) {
  const db = admin.firestore();
  let batch = db.batch();
  
  let count = 0;
  for (const q of questions) {
    const ref = db.collection("tat_questions").doc(q.id);
    batch.set(ref, q);
    
    if (++count % 500 === 0) {
      await batch.commit();
      batch = db.batch(); // Reset batch
    }
  }
  
  if (count % 500 !== 0) {
    await batch.commit();
  }
  
  return count;
}
```

**Validation Checklist:**
- ✅ All required fields present (id, text, options, correctAnswerId)
- ✅ Correct answer is in options list (no orphaned references)
- ✅ No duplicate IDs
- ✅ Image paths resolve (if embedded)
- ✅ Text not truncated (length check)
- ✅ Special characters properly encoded (unicode)

## Firestore Batch Operations (500-Doc Limit)

**Pattern:**
```javascript
const admin = require("firebase-admin");
const db = admin.firestore();

async function batchUpload(questions, collectionName = "tat_questions") {
  console.log(`Uploading ${questions.length} questions...`);
  
  let batch = db.batch();
  let batchCount = 0;
  let uploadedCount = 0;
  
  for (const question of questions) {
    const ref = db.collection(collectionName).doc(question.id);
    
    batch.set(ref, {
      ...question,
      uploadedAt: admin.firestore.FieldValue.serverTimestamp(),
      source: "batch_upload_v2"
    });
    
    batchCount++;
    
    // Firestore limit: 500 writes per batch
    if (batchCount === 500) {
      try {
        await batch.commit();
        uploadedCount += batchCount;
        console.log(`✓ Uploaded ${uploadedCount}/${questions.length}`);
        batch = db.batch();
        batchCount = 0;
      } catch (error) {
        console.error(`✗ Batch upload failed at ${uploadedCount}:`, error);
        throw error;
      }
    }
  }
  
  // Final batch
  if (batchCount > 0) {
    try {
      await batch.commit();
      uploadedCount += batchCount;
      console.log(`✓ Uploaded ${uploadedCount}/${questions.length}`);
    } catch (error) {
      console.error(`✗ Final batch upload failed:`, error);
      throw error;
    }
  }
  
  console.log(`\n✅ SUCCESS: Uploaded ${uploadedCount} questions`);
  return uploadedCount;
}

// Usage
const questions = require("./tat_batch_001.json");
batchUpload(questions)
  .catch(err => process.exit(1));
```

## Error Recovery & Idempotency

**Pattern: Resume from last successful checkpoint**

```javascript
async function resumableUpload(questions, checkpointFile = ".upload_checkpoint") {
  let startIndex = 0;
  let uploaded = 0;
  
  // Check for checkpoint (resume interrupted uploads)
  try {
    const checkpoint = JSON.parse(
      require("fs").readFileSync(checkpointFile, "utf8")
    );
    startIndex = checkpoint.lastIndex + 1;
    uploaded = checkpoint.uploadedCount;
    console.log(`Resuming from question ${startIndex} (${uploaded} already uploaded)`);
  } catch (e) {
    // No checkpoint, starting fresh
  }
  
  let batch = db.batch();
  let batchCount = 0;
  
  for (let i = startIndex; i < questions.length; i++) {
    const question = questions[i];
    
    batch.set(
      db.collection("tat_questions").doc(question.id),
      question
    );
    
    batchCount++;
    
    if (batchCount === 500 || i === questions.length - 1) {
      try {
        await batch.commit();
        uploaded += batchCount;
        batch = db.batch();
        batchCount = 0;
        
        // Save checkpoint (idempotent: safe to re-run)
        require("fs").writeFileSync(
          checkpointFile,
          JSON.stringify({ lastIndex: i, uploadedCount: uploaded })
        );
        
        console.log(`✓ Checkpoint: ${i + 1}/${questions.length}`);
      } catch (error) {
        console.error(`✗ Upload failed at ${i}. Rerun script to resume.`);
        throw error;
      }
    }
  }
  
  // Clean up checkpoint on success
  require("fs").unlinkSync(checkpointFile);
  console.log(`✅ Completed: ${uploaded} questions uploaded`);
}
```

**Idempotency Principle:**
- ✅ If script runs twice, same result (no duplicate data)
- ✅ Safe to interrupt and resume
- ✅ Each document has `id` (duplicate detection)

## Data Quality Checks (Runtime)

**Pattern: Assert data integrity**

```python
def check_data_integrity(questions):
    """Verify no data corruption before Firestore upload"""
    
    # Check 1: No duplicate IDs
    ids = [q['id'] for q in questions]
    assert len(ids) == len(set(ids)), "Duplicate question IDs"
    
    # Check 2: All options have IDs
    for q in questions:
        option_ids = [opt['id'] for opt in q['options']]
        assert len(option_ids) == len(set(option_ids)), f"Q{q['id']}: Duplicate option IDs"
    
    # Check 3: CorrectAnswerId is in options
    for q in questions:
        valid_ids = [opt['id'] for opt in q['options']]
        assert q['correctAnswerId'] in valid_ids, f"Q{q['id']}: correctAnswerId not in options"
    
    # Check 4: Text encoding (no truncation)
    for q in questions:
        assert len(q['questionText']) > 10, f"Q{q['id']}: Question too short (truncated?)"
        for opt in q['options']:
            assert len(opt['text']) > 0, f"Q{q['id']}: Option empty"
    
    # Check 5: Image files exist (if applicable)
    for q in questions:
        if 'imageUrl' in q and q['imageUrl']:
            assert os.path.exists(q['imageUrl']), f"Q{q['id']}: Image not found at {q['imageUrl']}"
    
    print(f"✅ All {len(questions)} questions passed integrity checks")
    return True

# Usage
check_data_integrity(questions)
uploaded_count = upload_to_firestore(questions)
```

## LLM Enrichment (Safe Usage)

**Pattern: LLM for metadata only (never correctness)**

```javascript
async function enrichWithGemini(questions) {
  const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);
  const model = genAI.getGenerativeModel({ model: "gemini-2.0-flash" });
  
  for (const question of questions) {
    // Only enrich metadata, NOT answer keys
    const prompt = `For this question, provide JSON metadata (do NOT modify the question or answers):
    
Question: ${question.questionText}

Provide ONLY this JSON (no other text):
{
  "difficulty": "easy|medium|hard",
  "category": "strategy|time_management|verbal",
  "keywords": ["key1", "key2"]
}`;
    
    try {
      const response = await model.generateContent(prompt);
      const metadata = JSON.parse(response.response.text());
      
      // Enrich question with metadata (never overwrite answer data)
      question.metadata = metadata;
    } catch (error) {
      console.warn(`Enrichment skipped for Q${question.id}: ${error}`);
      // Continue (enrichment is optional)
    }
  }
  
  return questions;
}

// Usage
const extracted = extractFromPDF(pdf_file);            // Deterministic
const validation = validateQuestions(extracted);       // Human review
if (!validation.isValid) throw new Error(validation.errors.join('\n'));
const enriched = await enrichWithGemini(extracted);    // Optional metadata
await uploadToFirestore(enriched);                     // Final upload
```

## Best Practices

1. **Extract deterministically** — use text layer parsing, not vision models
2. **LLM for enrichment only** — never for correctness
3. **Validate before upload** — catch errors early
4. **HTML preview for humans** — manual gate before final upload
5. **Batch operations** — respect 500-doc limit
6. **Checkpoint progress** — resume on failure
7. **Idempotency** — safe to run scripts multiple times
8. **Document format** — create format_spec.md for each data type
